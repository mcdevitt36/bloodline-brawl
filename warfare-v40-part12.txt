function moveActorHorizontal(a,dx,dz){
  const p=a.mesh.position.clone();p.x+=dx;p.z+=dz;const radius=.62;
  p.x=clamp(p.x,-59+radius,59-radius);p.z=clamp(p.z,-59+radius,59-radius);
  for(const o of obstacles){
    const nx=clamp(p.x,o.minX,o.maxX),nz=clamp(p.z,o.minZ,o.maxZ),vx=p.x-nx,vz=p.z-nz,d2=vx*vx+vz*vz;
    if(d2<radius*radius){if(d2>.0001){const d=Math.sqrt(d2),push=radius-d;p.x+=vx/d*push;p.z+=vz/d*push;}else{const left=Math.abs(p.x-o.minX),right=Math.abs(o.maxX-p.x),top=Math.abs(p.z-o.minZ),bottom=Math.abs(o.maxZ-p.z),m=Math.min(left,right,top,bottom);if(m===left)p.x=o.minX-radius;else if(m===right)p.x=o.maxX+radius;else if(m===top)p.z=o.minZ-radius;else p.z=o.maxZ+radius;}}
  }
  a.mesh.position.x=p.x;a.mesh.position.z=p.z;
}
