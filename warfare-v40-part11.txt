    else{
      const s=Math.ceil(left/1000);if(s!==state.lastCountdownSecond){state.lastCountdownSecond=s;countdownNumber.textContent=String(s);if(s<=3)countdown.classList.add('urgent');else countdown.classList.remove('urgent');}
      matchTimer.textContent='5:00';

  }
}
