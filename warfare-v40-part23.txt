const hubPortals=[];
let hubEnteredAt=0;

function buildHubWorldV42(){
  const g=new THREE.Group();g.name='WARFARE_V42_HUB';scene.add(g);
  // Main coastal staging plaza.
  addBox(g,0,-.5,0,104,1,104,0x31483f,{shoot:true,rough:.92});
  addBox(g,0,-.42,-38,104,.18,28,0xb7a574,{shoot:true,rough:.96});
  addBox(g,0,-.48,-51,104,.12,8,0x1b6486,{shoot:false,rough:.4});
  // Boardwalk spine.
  addBox(g,0,.03,-26,72,.22,9,0x87664a,{shoot:true,rough:.88});
  for(let x=-34;x<=34;x+=4)addBox(g,x,.18,-26,.12,.08,8.6,0x584331,{shoot:false,rough:.9});
  // Central family staging pavilion.
  addBox(g,0,.25,10,24,.5,16,0x263840,{shoot:true,rough:.82});
  addBox(g,0,2.15,17,18,4.3,2.0,0x1f2d36,{collide:true,rough:.8});
  addBox(g,-10,1.4,10,2,2.8,14,0x344852,{collide:true});
  addBox(g,10,1.4,10,2,2.8,14,0x344852,{collide:true});
  // Fountain / landmark.
  addCylinder(g,0,.2,8,4.8,4.8,.42,0x536775,{collide:true,metal:.12});
  addCylinder(g,0,.65,8,2.9,3.4,.9,0x2c6c84,{shoot:false,metal:.05});
  addCylinder(g,0,2.15,8,.36,.5,3.2,0x657781,{shoot:false,metal:.16});
  addSphere(g,0,4.05,8,.68,0x86d8ff,{emissive:0x194f66,shoot:false});
  const fountainLight=new THREE.PointLight(0x78cfff,2.8,14,2);fountainLight.position.set(0,4,8);g.add(fountainLight);
  // Landscaping and cover-like hub dressing.
  for(const [x,z] of [[-36,23],[-25,32],[25,32],[36,23],[-38,-7],[38,-7]]){
    addBox(g,x,.45,z,5,.9,3,0x4a5351,{collide:true,rough:.9});
    for(let i=-1;i<=1;i++)addSphere(g,x+i*1.2,1.35,z,.75,0x2e744e,{shoot:false,rough:.94});
  }
  // Lamps along boardwalk/plaza.
  for(const x of [-32,-20,-8,8,20,32]){
    addCylinder(g,x,1.6,-20,.08,.1,3.2,0x313b43,{collide:true,metal:.38});
    addSphere(g,x,3.25,-20,.18,0xffd18b,{emissive:0xffa640,shoot:false});
    const l=new THREE.PointLight(0xffc777,1.8,9,2);l.position.set(x,3.2,-20);g.add(l);
  }
  // Battlefield gates: deliberately large, readable, and impossible to confuse.
  function portal(map,name,x,z,color){
    const group=new THREE.Group();group.name=`V42_PORTAL_${map.toUpperCase()}`;group.position.set(x,0,z);g.add(group);
    addBox(group,0,.18,0,11,.36,7,0x182630,{shoot:false,rough:.7});
    addBox(group,-5.1,3.1,0,.7,6.2,.8,0x2b3944,{shoot:false,metal:.24});
    addBox(group,5.1,3.1,0,.7,6.2,.8,0x2b3944,{shoot:false,metal:.24});
    addBox(group,0,6.0,0,10.9,.7,.8,0x2b3944,{shoot:false,metal:.24});
    const glow=addBox(group,0,3.15,.16,9.4,5.1,.12,color,{shoot:false,rough:.3,metal:.05});glow.material.transparent=true;glow.material.opacity=.20;glow.material.emissive.setHex(color);glow.material.emissiveIntensity=1.35;
    const ring=new THREE.Mesh(new THREE.TorusGeometry(3.1,.12,10,48),new THREE.MeshBasicMaterial({color,transparent:true,opacity:.92}));ring.position.set(0,3.15,.35);group.add(ring);
    const floor=new THREE.Mesh(new THREE.RingGeometry(2.3,3.6,40),new THREE.MeshBasicMaterial({color,side:THREE.DoubleSide,transparent:true,opacity:.5}));floor.rotation.x=-Math.PI/2;floor.position.y=.39;group.add(floor);
    const light=new THREE.PointLight(color,4.8,18,2);light.position.set(0,3.3,2);group.add(light);
    hubPortals.push({map,name,group,ring,floor,worldPos:new THREE.Vector3(x,0,z)});
  }
  portal('city','CITY ROOFTOP',-24,-7,0x52bfff);
  portal('haunted','HAUNTED BEACH HOUSE',24,-7,0xb678ff);
  // Outer boundary as attractive railing rather than invisible walls.
  addBox(g,0,.75,51,104,1.5,.45,0x4e5e62,{collide:true,metal:.28});
  addBox(g,-51,.75,0,.45,1.5,102,0x4e5e62,{collide:true,metal:.28});
  addBox(g,51,.75,0,.45,1.5,102,0x4e5e62,{collide:true,metal:.28});
  // Hub-specific warm/cool lighting balance.
  const rim=new THREE.DirectionalLight(0xffd0a0,.75);rim.position.set(-25,28,18);scene.add(rim);
}

const legacyResetSceneV42=resetScene;
resetScene=function(map){
  if(map!=='hub')return legacyResetSceneV42(map);
  scene=new THREE.Scene();scene.background=new THREE.Color(0x10283b);scene.fog=new THREE.Fog(0x183447,70,180);
  actors=[];pickups=[];containers=[];obstacles=[];worldMeshes=[];effects=[];player=null;hubPortals.length=0;
  const hemi=new THREE.HemisphereLight(0xcbe9ff,0x26362e,1.45);scene.add(hemi);
  const sun=new THREE.DirectionalLight(0xffe7c9,1.85);sun.position.set(34,52,24);sun.castShadow=true;sun.shadow.mapSize.set(2048,2048);sun.shadow.camera.left=-70;sun.shadow.camera.right=70;sun.shadow.camera.top=70;sun.shadow.camera.bottom=-70;scene.add(sun);
  buildHubWorldV42();
};

function enterHubV42(){
  initRenderer();
  state.phase='hub';state.spectating=false;state.playerEliminated=false;state.lastCountdownSecond=null;hubEnteredAt=now();
  boot.classList.add('hidden');eliminated.classList.add('hidden');resultScreen.classList.add('hidden');countdown.classList.add('hidden');spectateBanner.classList.add('hidden');hud.classList.remove('hidden');
  resetScene('hub');
  player=makeActor(state.selectedCharacter,new THREE.Vector3(0,0,33),false,CHARACTERS[state.selectedCharacter].name);player.heading=Math.PI;player.mesh.rotation.y=Math.PI;
  cameraYaw=Math.PI;cameraPitch=-.14;
  hudName.textContent=player.name;modeLabel.textContent='WARFARE HUB';matchTimer.textContent='HUB';aliveText.textContent='CHOOSE BATTLEFIELD';weaponName.textContent='UNARMED';ammoText.textContent=CHARACTERS[player.charId].melee;
  feed.innerHTML='';addFeed('Welcome to the Warfare Hub. Walk to a battlefield gate and press E.');
  updateLoadoutVisual(player);updateHUD();
  try{canvas.requestPointerLock?.();}catch(_){ }
}

function nearestHubPortalV42(max=5.2){
  if(!player||state.phase!=='hub')return null;let best=null,bd=max;
  for(const p of hubPortals){const d=player.mesh.position.distanceTo(p.worldPos);if(d<bd){bd=d;best=p;}}
  return best;
}
function activateHubPortalV42(){
  const p=nearestHubPortalV42(5.2);if(!p)return false;
  state.selectedMap=p.map;addFeed(`${p.name} selected — deploying.`);playTone(720,.12,.07,'sine');
  startMatch();return true;
}

// Hub-specific interaction prompt overrides battle pickup prompts only while in hub.
const legacyInteractionPromptV42=updateInteractionPrompt;
updateInteractionPrompt=function(){
  if(state.phase!=='hub')return legacyInteractionPromptV42();
  if(!player){interactionPrompt.classList.add('hidden');return;}
  const p=nearestHubPortalV42(6.4);
  if(p){interactionPrompt.textContent=`${p.name} • E — DEPLOY`;interactionPrompt.classList.remove('hidden');}
  else{interactionPrompt.textContent='EXPLORE THE HUB • FIND A BATTLEFIELD GATE';interactionPrompt.classList.remove('hidden');}
};

const legacyHUDV42=updateHUD;
updateHUD=function(){
  legacyHUDV42();
  if(state.phase==='hub'){
    matchTimer.textContent='HUB';aliveText.textContent='CHOOSE BATTLEFIELD';statsKills.textContent='0';statsDamage.textContent='0';
    if(player&&!currentWeapon(player)){weaponName.textContent='UNARMED';ammoText.textContent=CHARACTERS[player.charId].melee;}
  }
};

const legacyPickupsV42=updatePickups;
updatePickups=function(t){
  legacyPickupsV42(t);
  if(state.phase==='hub')for(const [i,p] of hubPortals.entries()){
    p.ring.rotation.z=t*.00055*(i? -1:1);p.floor.rotation.z=t*.00028*(i?1:-1);
    const pulse=.74+Math.sin(t*.003+i)*.18;p.ring.material.opacity=pulse;
  }
};

// The flat menu now selects only the fighter. The 3D hub selects the battlefield.
mapChoices.style.display='none';
const mapTitle=mapChoices.previousElementSibling;if(mapTitle&&mapTitle.classList.contains('menu-section-title'))mapTitle.style.display='none';
playButton.textContent='ENTER WARFARE HUB';

// Capture phase prevents the old direct startMatch button binding from firing.
playButton.addEventListener('click',e=>{e.preventDefault();e.stopImmediatePropagation();enterHubV42();},true);
returnWarfareButton.addEventListener('click',e=>{e.preventDefault();e.stopImmediatePropagation();enterHubV42();},true);
resultReturn.addEventListener('click',e=>{e.preventDefault();e.stopImmediatePropagation();enterHubV42();},true);

// In-hub E activates the nearest battlefield gate instead of battle loot interaction.
addEventListener('keydown',e=>{
  if(state.phase==='hub'&&e.code==='KeyE'){
    e.preventDefault();e.stopImmediatePropagation();activateHubPortalV42();
  }
},true);
