function updateCamera(dt){
  const subject=state.spectating?spectateTarget():player;if(!subject)return;
  const target=subject.mesh.position.clone().add(new THREE.Vector3(0,subject.crouching?1.5:1.9,0));
  if(state.spectating){const f=actorForward(subject);cameraYaw=Math.atan2(-f.x,-f.z);}
  const dist=state.spectating?6.8:(aiming?3.7:5.9);const shoulder=state.spectating?0:aiming?.85:.5;
  const back=new THREE.Vector3(Math.sin(cameraYaw)*dist,1.35-Math.sin(cameraPitch)*2.1,Math.cos(cameraYaw)*dist);const right=new THREE.Vector3(Math.cos(cameraYaw),0,-Math.sin(cameraYaw)).multiplyScalar(shoulder);let desired=target.clone().add(back).add(right);
  const dir=desired.clone().sub(target),len=dir.length();dir.normalize();raycaster.set(target,dir);raycaster.far=len;const hit=raycaster.intersectObjects(worldMeshes,false)[0];if(hit&&hit.distance<len)desired=target.clone().addScaledVector(dir,Math.max(.65,hit.distance-.28));
  camera.position.lerp(desired,1-Math.exp(-10*dt));camera.lookAt(target.clone().add(new THREE.Vector3(0,Math.tan(cameraPitch)*2.2,0)));
  const targetFov=state.spectating?70:(aiming?57:((keys.has('ShiftLeft')||keys.has('ShiftRight'))?76:68));camera.fov=THREE.MathUtils.damp(camera.fov,targetFov,8,dt);camera.updateProjectionMatrix();
}
