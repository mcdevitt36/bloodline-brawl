function interact(){
  if(!player||player.dead||state.spectating)return;
  let best=null,bd=2.6;
  for(const p of pickups){if(p.taken)continue;const d=player.mesh.position.distanceTo(p.group.position);if(d<bd){bd=d;best={type:'pickup',obj:p};}}
  for(const c of containers){if(c.opened)continue;const d=player.mesh.position.distanceTo(c.group.position);if(d<bd){bd=d;best={type:'container',obj:c};}}
  if(!best)return;
  if(best.type==='pickup')takePickup(player,best.obj);else openContainer(player,best.obj);
}

}
