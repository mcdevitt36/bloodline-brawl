function meleeTargetV46(range){

    if(a===player||a.dead)continue;

      [a.mesh.position.clone().add(new THREE.Vector3(0,1.28,0)),.78],
      [a.mesh.position.clone().add(new THREE.Vector3(0,2.38,0)),.48]
    ];

      const d=raySphereDistance(origin,dir,center,r);
      if(d!==null&&d<bestDist){best=a;bestDist=d;bestPoint=origin.clone().addScaledVector(dir,d);}
    }

  if(wall&&wall.distance<bestDist-.08)return null;
  return{target:best,point:bestPoint||best.mesh.position.clone().add(new THREE.Vector3(0,1.3,0)),distance:bestDist};
}
