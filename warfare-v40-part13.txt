function wanderBot(bot,dt,t){if(t>bot.thinkAt){bot.thinkAt=t+rand(900,1800);bot.heading+=rand(-1.3,1.3);}moveBotToward(bot,new THREE.Vector3(-Math.sin(bot.heading),0,-Math.cos(bot.heading)),dt,5.5);bot.mesh.rotation.y=bot.heading;}
